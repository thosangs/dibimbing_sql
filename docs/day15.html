// 01 — Subqueries

Scalar, Correlated & Set Subqueries

Subquery adalah query di dalam query. Scalar subquery mengembalikan tepat satu nilai (satu baris, satu kolom) dan bisa dipakai di mana saja ekspresi diizinkan. Subquery dengan IN mengembalikan daftar nilai — PostgreSQL mengonversinya ke hash semi-join. Correlated subquery mereferensikan tabel dari query luar — dievaluasi ulang per-baris sehingga lebih lambat; pertimbangkan JOIN atau CTE sebagai alternatif.

examples/subqueries.sql
-- IN subquery: customer yang pernah menyewa 'ACADEMY DINOSAUR'
SELECT DISTINCT c.customer_id, c.first_name, c.last_name
FROM customer c
WHERE c.customer_id IN (
  SELECT r.customer_id
  FROM rental r
  JOIN inventory i ON i.inventory_id = r.inventory_id
  JOIN film f      ON f.film_id = i.film_id
  WHERE UPPER(f.title) = 'ACADEMY DINOSAUR'
);

-- Nested IN: film dengan actor yang main di lebih dari 10 film
SELECT f.film_id, f.title
FROM film f
WHERE f.film_id IN (
  SELECT fa.film_id
  FROM film_actor fa
  WHERE fa.actor_id IN (
    SELECT actor_id
    FROM film_actor
    GROUP BY actor_id
    HAVING COUNT(*) > 10   -- filter actor produktif
  )
);

-- Subquery di FROM (derived table): top 5 actor by film count
SELECT a.actor_id, a.first_name, a.last_name, actor_stats.film_count
FROM (
  SELECT fa.actor_id, COUNT(*) AS film_count
  FROM film_actor fa
  GROUP BY fa.actor_id
) AS actor_stats
JOIN actor a ON a.actor_id = actor_stats.actor_id
ORDER BY actor_stats.film_count DESC, a.actor_id
LIMIT 5;

Subquery di FROM disebut derived table — harus selalu diberi alias. Berbeda dengan CTE, derived table tidak bisa direferensikan lebih dari sekali dalam satu query.

// 02 — CTEs

Common Table Expressions (WITH Clause)

CTE mendefinisikan named temporary result set di awal query menggunakan WITH name AS (...). CTE membuat query kompleks lebih mudah dibaca dengan memecahnya menjadi langkah-langkah bernama. Secara default PostgreSQL memperlakukan CTE sebagai "optimization fence" hanya sampai v12 — sejak v12 planner bisa inline CTE jika menguntungkan. Multiple CTEs bisa didefinisikan dalam satu blok WITH dipisah koma dan saling mereferensikan.

examples/ctes.sql
-- CTE sederhana: top 10 customer by total spend
WITH customer_spend AS (
  SELECT customer_id, SUM(amount) AS total_spend
  FROM payment
  GROUP BY customer_id
)
SELECT c.customer_id, c.first_name, c.last_name, cs.total_spend
FROM customer c
JOIN customer_spend cs USING (customer_id)
ORDER BY cs.total_spend DESC
LIMIT 10;

-- CTE dengan window function: rental ke-2 tiap customer
WITH ranked_rentals AS (
  SELECT customer_id, rental_id, rental_date,
         ROW_NUMBER() OVER (PARTITION BY customer_id
                            ORDER BY rental_date DESC) AS rn
  FROM rental
)
SELECT customer_id, rental_id, rental_date
FROM ranked_rentals
WHERE rn = 2    -- filter SETELAH ranking dihitung
ORDER BY customer_id;

CTE sangat berguna dikombinasikan dengan window functions karena WHERE tidak bisa dipakai langsung pada hasil window function — harus dibungkus dulu dalam subquery atau CTE, lalu difilter di luar.

// 03 — Ranking Functions

ROW_NUMBER, RANK, DENSE_RANK

Tiga fungsi ranking memberikan nomor urut per grup. ROW_NUMBER() selalu unik — tidak ada dua baris yang dapat nilai sama meskipun tiebreak. RANK() memberikan nomor sama untuk tie tapi melewati angka berikutnya (1, 2, 2, 4). DENSE_RANK() memberikan nomor sama untuk tie tanpa melewati angka (1, 2, 2, 3). Semua membutuhkan OVER (ORDER BY ...) dan opsional PARTITION BY untuk ranking per grup.

// ROW_NUMBER vs RANK vs DENSE_RANK

Film        | Rentals | ROW_NUMBER | RANK | DENSE_RANK
────────────┼─────────┼────────────┼──────┼───────────
Bucket Brother│  34   │     1      │  1   │     1
Rocketeer    │  33   │     2      │  2   │     2
Ridgemont    │  33   │     3      │  2   │     2    ← tie
Juggler      │  32   │     4      │  4   │     3    ← RANK skip ke 4, DENSE_RANK lanjut ke 3
Zorro Ark    │  31   │     5      │  5   │     4
examples (CTE + ranking)
-- Rental ke-2 tiap customer (pakai ROW_NUMBER karena harus unik)
WITH ranked_rentals AS (
  SELECT customer_id, rental_id, rental_date,
         ROW_NUMBER() OVER (
           PARTITION BY customer_id
           ORDER BY rental_date DESC
         ) AS rn
  FROM rental
)
SELECT customer_id, rental_id, rental_date
FROM ranked_rentals
WHERE rn = 2
ORDER BY customer_id;

-- Top 3 film per kategori (RANK: boleh ada tie di posisi 1)
WITH film_rentals AS (
  SELECT c.name AS category, f.title,
         COUNT(r.rental_id) AS rentals
  FROM category c
  JOIN film_category fc ON fc.category_id = c.category_id
  JOIN film f           ON f.film_id = fc.film_id
  JOIN inventory i      ON i.film_id = f.film_id
  JOIN rental r         ON r.inventory_id = i.inventory_id
  GROUP BY c.name, f.title
),
ranked AS (
  SELECT *, DENSE_RANK() OVER (
    PARTITION BY category ORDER BY rentals DESC
  ) AS rnk
  FROM film_rentals
)
SELECT category, title, rentals, rnk
FROM ranked
WHERE rnk <= 3
ORDER BY category, rnk;
// 04 — String Functions

LEFT, SUBSTRING, CONCAT, UPPER, ILIKE

PostgreSQL memiliki library string yang kaya. LEFT/RIGHT mengekstrak n karakter dari ujung string. SUBSTRING (atau substr) mengekstrak bagian tengah. CONCAT menggabungkan string — alternatifnya operator ||. UPPER/LOWER mengubah case. ILIKE adalah case-insensitive LIKE — lebih lambat karena tidak bisa menggunakan B-tree index biasa (butuh citext atau index functional).

examples/strings.sql
-- LEFT: 5 karakter pertama dari judul film
SELECT title, LEFT(title, 5) AS left5
FROM film ORDER BY film_id LIMIT 10;

-- SUBSTRING: karakter posisi 3 sampai 8 (panjang 6)
SELECT title, SUBSTRING(title FROM 3 FOR 6) AS sub_3_8
FROM film ORDER BY film_id LIMIT 10;

-- REPLACE: ganti 'A' dengan '*'
SELECT title, REPLACE(title, 'A', '*') AS replaced
FROM film ORDER BY film_id LIMIT 10;

-- CONCAT: nama lengkap actor
SELECT CONCAT(first_name, ' ', last_name) AS actor_name
FROM actor ORDER BY actor_id LIMIT 10;

-- LENGTH: panjang judul (dalam bytes = chars untuk ASCII)
SELECT title, LENGTH(title) AS title_len
FROM film ORDER BY title_len DESC, title LIMIT 10;

-- ILIKE: case-insensitive match
SELECT title FROM film
WHERE title ILIKE 'aca%'   -- 'ACADEMY...', 'academy...', 'Aca...'
ORDER BY film_id LIMIT 10;
// 05 — Date & Time Functions

NOW, DATE_TRUNC, EXTRACT, INTERVAL

DATE_TRUNC memotong timestamp ke granularitas tertentu (hari, bulan, tahun) — hasilnya tetap TIMESTAMP, bukan DATE, sehingga bisa langsung di-GROUP BY untuk aggregasi per-bulan. EXTRACT mengambil field numerik (jam, hari, bulan, tahun). INTERVAL merepresentasikan durasi dan bisa dijumlahkan/dikurangkan dari TIMESTAMP. Untuk timestamp column, gunakan half-open range (>= start AND < end) bukan BETWEEN agar index-friendly dan tidak ada boundary issue.

examples/datetime.sql
-- NOW dan interval arithmetic
SELECT NOW() AS now_ts,
       NOW() + INTERVAL '7 days'  AS in_7_days,
       NOW() - INTERVAL '3 hours' AS three_hours_ago;

-- DATE_TRUNC: payment per bulan (GROUP BY)
SELECT DATE_TRUNC('month', payment_date) AS month,
       SUM(amount) AS total_amount
FROM payment
GROUP BY 1
ORDER BY 1;

-- EXTRACT: distribusi payment per jam
SELECT EXTRACT(HOUR FROM payment_date) AS hour_of_day,
       COUNT(*) AS payments
FROM payment
GROUP BY 1
ORDER BY 1;

-- Half-open range: cara BENAR filter timestamp per hari
SELECT payment_id, payment_date, amount
FROM payment
WHERE payment_date >= DATE '2005-05-25'
  AND payment_date <  DATE '2005-05-26'  -- tidak inklusif akhir
ORDER BY payment_date LIMIT 10;

-- Relative window: 7 hari terakhir
SELECT COUNT(*) AS payments_last_7_days
FROM payment
WHERE payment_date >= NOW() - INTERVAL '7 days';

Hindari payment_date::date = '2005-05-25' pada kolom TIMESTAMPTZ berindeks — cast menghancurkan kemampuan planner menggunakan index. Pakai half-open range: >= date AND < date + 1.
