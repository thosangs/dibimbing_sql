// 01 — Konsep Dasar

Relational Model & ACID

Relational database menyimpan data dalam tabel (relasi) — kumpulan baris (tuple) dengan kolom (atribut) bertipe fixed. Primary key uniquely mengidentifikasi setiap baris; foreign key membuat link antar tabel tanpa duplikasi data. Semua operasi dibungkus transaksi ACID: Atomicity (semua-atau-tidak-sama-sekali), Consistency (state valid), Isolation (transaksi paralel tidak saling ganggu), Durability (commit = permanen).

ACID

  • Atomicity — semua atau rollback
  • Consistency — state valid sebelum/sesudah
  • Isolation — tidak ada dirty reads
  • Durability — commit = disimpan permanent

SQL vs NoSQL

  • SQL: schema ketat, JOIN kuat, ACID penuh
  • Key-Value: Redis — cache, session
  • Document: MongoDB — JSON flexible
  • Columnar: Cassandra — write heavy
// 02 — DDL

Data Definition Language

DDL mendefinisikan struktur database — bukan datanya. CREATE TABLE mendefinisikan kolom dan constraint; ALTER TABLE mengubah schema yang sudah ada (tambah kolom, tambah constraint) tanpa rewrite seluruh tabel di PostgreSQL. CREATE INDEX membuat secondary lookup structure — B-tree by default — sehingga query dengan WHERE pada kolom tersebut tidak perlu full scan.

examples/ddl.sql
-- CREATE TABLE dengan PK dan FK referencing dvdrental.customer
DROP TABLE IF EXISTS public.customer_note CASCADE;
CREATE TABLE public.customer_note (
    customer_id integer NOT NULL REFERENCES public.customer(customer_id),
    note        text    NOT NULL,
    created_at  timestamptz NOT NULL DEFAULT now(),
    PRIMARY KEY (customer_id, created_at)   -- composite PK
);

-- ALTER TABLE: tambah kolom dan CHECK constraint
ALTER TABLE public.film
    ADD COLUMN IF NOT EXISTS imdb_url text;
-- CHECK: hanya nilai yang match regex yang bisa disimpan
ALTER TABLE public.film
    ADD CONSTRAINT film_imdb_url_chk
    CHECK (imdb_url IS NULL OR imdb_url ~ '^https?://');

-- Partial index: hanya index rental yang belum dikembalikan
-- Index kecil, lookup cepat untuk query outstanding rentals
CREATE INDEX IF NOT EXISTS idx_rental_outstanding
    ON public.rental (rental_date)
    WHERE return_date IS NULL;

PRIMARY KEY (customer_id, created_at) — composite PK, kombinasi dua kolom harus unique. REFERENCES otomatis membuat FK constraint dengan integrity check. WHERE return_date IS NULL pada CREATE INDEX = partial index: hanya baris yang match kondisi yang diindeks, sehingga index jauh lebih kecil dan lebih cepat.

// 03 — DML

Data Manipulation Language

INSERT menambah baris baru. UPDATE mengubah nilai kolom pada baris yang match WHERE — tanpa WHERE, semua baris berubah. DELETE menghapus baris. UPSERT (INSERT ... ON CONFLICT DO UPDATE) adalah operasi atomik: insert kalau tidak ada konflik, update kalau ada — menghindari race condition antara SELECT + INSERT terpisah. RETURNING mengembalikan nilai kolom dari baris yang baru dimodifikasi tanpa SELECT tambahan.

examples/dml.sql
BEGIN;   -- semua operasi dibungkus transaksi

-- INSERT dengan RETURNING: langsung dapat actor_id yang di-generate
INSERT INTO public.actor (first_name, last_name)
VALUES ('TRAIN', 'DEMO')
RETURNING actor_id, first_name, last_name;

-- UPDATE: naikkan rental_rate untuk film G/PG
UPDATE public.film
SET rental_rate = rental_rate + 0.50
WHERE rating IN ('G', 'PG')
RETURNING film_id, title, rating, rental_rate;

-- DELETE dengan USING: referensikan tabel lain di klausa WHERE
DELETE FROM public.film_actor fa
USING public.actor a
WHERE a.actor_id = fa.actor_id
  AND a.first_name = 'TRAIN'
  AND a.last_name  = 'DEMO'
RETURNING fa.actor_id, fa.film_id;

-- UPSERT: ON CONFLICT mencegah error duplicate key
-- Jika actor_id=99999 sudah ada → DO UPDATE; kalau tidak → INSERT
INSERT INTO public.actor (actor_id, first_name, last_name)
VALUES (99999, 'UPSERT', 'DEMO')
ON CONFLICT (actor_id) DO UPDATE
SET first_name  = EXCLUDED.first_name,
    last_name   = EXCLUDED.last_name,
    last_update = now()
RETURNING actor_id, first_name, last_name;

ROLLBACK;  -- demo saja, kembalikan semua perubahan

EXCLUDED adalah pseudo-table yang merujuk ke baris yang gagal di-insert karena conflict — sehingga kita bisa akses nilai yang ingin di-upsert. RETURNING bekerja seperti SELECT pada baris yang baru dimodifikasi, tanpa round-trip SQL tambahan.

// 04 — Column Types

Tipe Data PostgreSQL

PostgreSQL memiliki type system yang kaya. CHAR(n) selalu disimpan n karakter (space-padded) — jarang dipakai. VARCHAR(n) variabel sampai n, TEXT tanpa batas. NUMERIC(p,s) fixed-precision cocok untuk uang — tidak ada floating-point error seperti FLOAT. TIMESTAMPTZ menyimpan timezone offset sehingga aman lintas zona waktu; DATE hanya tanggal. INTERVAL merepresentasikan durasi waktu dan bisa diaritmetikkan langsung dengan DATE/TIMESTAMP.

examples/column_types.sql
CREATE TABLE public.typing_demo (
    id          serial       PRIMARY KEY,
    code        char(5),              -- fixed 5 chars, space-padded jika lebih pendek
    name        varchar(50),          -- max 50 chars, tidak di-pad
    description text,                 -- unlimited
    active      boolean DEFAULT true, -- true/false/'t'/'f'/1/0
    qty         integer,              -- 32-bit signed int
    price       numeric(10,2),        -- 10 digit total, 2 di belakang desimal
    created_on  date DEFAULT current_date,
    updated_at  timestamptz DEFAULT now(),
    duration    interval              -- e.g. '3 days 4 hours', '90 minutes'
);

-- Demonstrasi arithmetic dengan types
SELECT
    id,
    code,
    length(code)              AS len_code_char,   -- CHAR: hitung padding
    rtrim(code)               AS code_trimmed,    -- trim trailing spaces
    price,
    price::text               AS price_as_text,   -- cast numeric → text
    (created_on + duration)   AS created_plus_duration  -- date + interval
FROM public.typing_demo
ORDER BY id;

CHAR(5) untuk nilai 'AB' akan disimpan sebagai 'AB ' (3 spasi) — length() mengembalikan 5, bukan 2. Gunakan rtrim() untuk membersihkannya. price::text adalah cast sintaks PostgreSQL; alternatifnya CAST(price AS text). created_on + duration bekerja karena PostgreSQL tahu cara menjumlahkan DATE + INTERVAL.

// 05 — Materialization Strategies

CTE, View, Materialized View, Temp Table

PostgreSQL menawarkan beberapa cara menyimpan/menstrukturisasi hasil query sementara atau permanen. CTE (WITH) adalah subquery bernama dalam satu statement — biasanya di-inline oleh planner, tapi berguna untuk readability. View adalah query yang disimpan tapi dieksekusi tiap kali dipanggil. Materialized View menyimpan hasilnya secara fisik — perlu REFRESH manual untuk update data. Temp Table hanya ada selama session berlangsung.

examples/materialization.sql
-- 1) CTE: query bernama dalam satu statement
WITH recent_rentals AS (
    SELECT r.rental_id, r.customer_id, r.rental_date
    FROM public.rental r
    WHERE r.rental_date >= now() - interval '60 days'
)
SELECT c.customer_id, c.first_name, c.last_name,
       COUNT(rr.rental_id) AS rentals_last_60d
FROM public.customer c
LEFT JOIN recent_rentals rr ON rr.customer_id = c.customer_id
GROUP BY c.customer_id, c.first_name, c.last_name
ORDER BY rentals_last_60d DESC
LIMIT 10;

-- 2) View: virtual, dieksekusi setiap kali SELECT dipanggil
CREATE VIEW public.vw_film_rental_counts AS
SELECT f.film_id, f.title, x.num_rentals
FROM public.film f
LEFT JOIN (
    SELECT i.film_id, COUNT(r.rental_id) AS num_rentals
    FROM public.inventory i
    LEFT JOIN public.rental r ON r.inventory_id = i.inventory_id
    GROUP BY i.film_id
) x ON x.film_id = f.film_id;

-- 3) Materialized View: disimpan fisik, perlu REFRESH
CREATE MATERIALIZED VIEW public.mv_top_films AS
SELECT f.film_id, f.title, x.num_rentals
FROM public.film f
JOIN (
    SELECT i.film_id, COUNT(r.rental_id) AS num_rentals
    FROM public.inventory i
    JOIN public.rental r ON r.inventory_id = i.inventory_id
    GROUP BY i.film_id
) x ON x.film_id = f.film_id
ORDER BY x.num_rentals DESC
WITH NO DATA;  -- definisi dulu, data belakangan

REFRESH MATERIALIZED VIEW public.mv_top_films;  -- populasi data

// Perbandingan Strategi

Strategi          | Storage | Stale? | Use Case
──────────────────┼─────────┼────────┼──────────────────────────────
CTE (WITH)        | None    | No     | Readability, intermediate step
Derived Table     | None    | No     | One-off subquery in FROM
Temp Table        | Disk    | No     | Reuse result dalam satu session
UNLOGGED Table    | Disk    | No     | Staging data, crash-ok
View              | None    | No     | Abstraksi, access control
Materialized View | Disk    | YES    | Pre-compute heavy aggregations
